    <script>
        /**
         * RHUTHER OS - Android Physical Therapy OS Engine
         */
        const AndroidOS = {
            currentApp: null,
            openApps: [],
            soundEnabled: true,
            audioCtx: null,

            init() {
                this.updateClock();
                setInterval(() => this.updateClock(), 1000);
                lucide.createIcons();
                this.renderDrawerList();
                this.showToast("RHUTHER OS Android", "Sistema ativado! Pronto para os estudos de fisioterapia.");

                // Audio context initiation on user interaction
                window.addEventListener('touchstart', () => this.initAudio(), { once: true });
                window.addEventListener('click', () => this.initAudio(), { once: true });
            },

            initAudio() {
                if (!this.audioCtx) {
                    this.audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                }
            },

            playBeep(freq = 500) {
                if (!this.soundEnabled || !this.audioCtx) return;
                try {
                    const osc = this.audioCtx.createOscillator();
                    const gain = this.audioCtx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(freq, this.audioCtx.currentTime);
                    gain.gain.setValueAtTime(0.04, this.audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, this.audioCtx.currentTime + 0.08);
                    osc.connect(gain);
                    gain.connect(this.audioCtx.destination);
                    osc.start();
                    osc.stop(this.audioCtx.currentTime + 0.08);
                } catch(e) {}
            },

            updateClock() {
                const now = new Date();
                const timeStr = now.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
                const dateStr = now.toLocaleDateString('pt-BR', { weekday: 'long', day: '2-digit', month: 'long' });

                document.getElementById('android-clock').innerText = timeStr;
                document.getElementById('widget-time').innerText = timeStr;
                document.getElementById('widget-date').innerText = dateStr.charAt(0).toUpperCase() + dateStr.slice(1);
                document.getElementById('shade-date').innerText = dateStr;
            },

            toggleShade() {
                const shade = document.getElementById('quick-shade');
                shade.classList.toggle('hidden');
                this.playBeep(600);
            },

            setBrightness(val) {
                document.getElementById('phone-frame').style.filter = `brightness(${val}%)`;
            },

            toggleSound() {
                this.soundEnabled = !this.soundEnabled;
                const btn = document.getElementById('shade-sound-btn');
                btn.className = `quick-toggle p-3 rounded-2xl flex flex-col items-center justify-center space-y-1 transition active:scale-95 ${this.soundEnabled ? 'bg-teal-500 text-slate-950' : 'bg-slate-800 text-rose-400'}`;
                this.showToast("Áudio", this.soundEnabled ? "Som ativado" : "Som desativado");
            },

            toggleToggle(name, btn) {
                btn.classList.toggle('bg-teal-500');
                btn.classList.toggle('text-slate-950');
                btn.classList.toggle('bg-slate-800');
                btn.classList.toggle('text-slate-300');
                this.playBeep(700);
            },

            toggleDrawer() {
                const drawer = document.getElementById('app-drawer');
                drawer.classList.toggle('hidden');
                this.playBeep(550);
            },

            renderDrawerList() {
                const grid = document.getElementById('drawer-grid');
                if (!grid) return;
                grid.innerHTML = '';

                Object.keys(this.apps).forEach(appId => {
                    const app = this.apps[appId];
                    const btn = document.createElement('button');
                    btn.className = 'flex flex-col items-center group active:scale-90 transition drawer-item';
                    btn.onclick = () => {
                        this.toggleDrawer();
                        this.openApp(appId);
                    };
                    btn.innerHTML = `
                        <div class="w-12 h-12 rounded-2xl bg-slate-800 border border-slate-700 flex items-center justify-center text-teal-400 shadow group-hover:scale-105 transition">
                            <i data-lucide="${app.icon}" class="w-6 h-6"></i>
                        </div>
                        <span class="text-[10px] font-medium text-slate-300 mt-1.5 leading-tight text-center">${app.title}</span>
                    `;
                    grid.appendChild(btn);
                });
                lucide.createIcons();
            },

            filterDrawer(query) {
                const q = query.toLowerCase();
                document.querySelectorAll('.drawer-item').forEach(item => {
                    const text = item.innerText.toLowerCase();
                    item.style.display = text.includes(q) ? 'flex' : 'none';
                });
            },

            openApp(appId) {
                this.playBeep(650);
                const appConfig = this.apps[appId];
                if (!appConfig) return;

                if (!this.openApps.includes(appId)) {
                    this.openApps.push(appId);
                }

                this.currentApp = appId;

                // Close overlays
                document.getElementById('quick-shade').classList.add('hidden');
                document.getElementById('recents-screen').classList.add('hidden');
                document.getElementById('app-drawer').classList.add('hidden');

                // Render App Window
                const win = document.getElementById('app-window');
                const titleEl = document.getElementById('app-title');
                const iconEl = document.getElementById('app-icon');
                const bodyEl = document.getElementById('app-body');

                titleEl.innerText = appConfig.title;
                iconEl.setAttribute('data-lucide', appConfig.icon);
                bodyEl.innerHTML = appConfig.render();

                win.classList.remove('hidden');
                lucide.createIcons();

                if (appConfig.onOpen) {
                    setTimeout(() => appConfig.onOpen(), 50);
                }
            },

            goHome() {
                this.playBeep(400);
                document.getElementById('app-window').classList.add('hidden');
                document.getElementById('recents-screen').classList.add('hidden');
                document.getElementById('app-drawer').classList.add('hidden');
                document.getElementById('quick-shade').classList.add('hidden');
                this.currentApp = null;
            },

            goBack() {
                this.goHome();
            },

            navBack() {
                if (!document.getElementById('app-window').classList.contains('hidden')) {
                    this.goHome();
                } else if (!document.getElementById('recents-screen').classList.contains('hidden')) {
                    document.getElementById('recents-screen').classList.add('hidden');
                } else if (!document.getElementById('app-drawer').classList.contains('hidden')) {
                    document.getElementById('app-drawer').classList.add('hidden');
                } else if (!document.getElementById('quick-shade').classList.contains('hidden')) {
                    document.getElementById('quick-shade').classList.add('hidden');
                } else {
                    this.playBeep(300);
                }
            },

            toggleRecents() {
                this.playBeep(500);
                const recents = document.getElementById('recents-screen');
                const isHidden = recents.classList.contains('hidden');

                if (isHidden) {
                    this.renderRecentsList();
                    recents.classList.remove('hidden');
                    document.getElementById('app-window').classList.add('hidden');
                    document.getElementById('app-drawer').classList.add('hidden');
                } else {
                    recents.classList.add('hidden');
                }
            },

            renderRecentsList() {
                const list = document.getElementById('recents-list');
                if (!list) return;
                list.innerHTML = '';

                if (this.openApps.length === 0) {
                    list.innerHTML = `<div class="text-slate-500 text-xs w-full text-center py-10">Nenhum app recente aberto</div>`;
                    return;
                }

                this.openApps.forEach(appId => {
                    const app = this.apps[appId];
                    const card = document.createElement('div');
                    card.className = 'w-48 h-64 bg-slate-900 border border-teal-500/30 rounded-3xl p-3 flex flex-col justify-between shrink-0 shadow-xl cursor-pointer hover:border-teal-400 transition';
                    card.onclick = () => this.openApp(appId);
                    card.innerHTML = `
                        <div class="flex justify-between items-center border-b border-white/10 pb-2">
                            <div class="flex items-center space-x-2">
                                <i data-lucide="${app.icon}" class="w-4 h-4 text-teal-300"></i>
                                <span class="font-bold text-xs text-slate-200 truncate w-24">${app.title}</span>
                            </div>
                            <button onclick="event.stopPropagation(); AndroidOS.closeApp('${appId}')" class="p-1 text-slate-400 hover:text-rose-400">

                            </button>
                        </div>

                            Toque para visualizar
                        </div>
                        <div class="text-[9px] text-teal-400 font-semibold text-center">EM EXECUÇÃO</div>
                    `;
                    list.appendChild(card);
                });
                lucide.createIcons();
            },

            closeApp(appId) {
                this.openApps = this.openApps.filter(id => id !== appId);
                if (this.currentApp === appId) {
                    this.goHome();
                }
                this.renderRecentsList();
            },

            clearAllApps() {
                this.openApps = [];
                this.goHome();
                this.showToast("Multitarefa", "Todos os aplicativos foram encerrados.");
            },

                res.innerText = 'Incorreto. Analise os testes de translação anterior de tíbia.';
                this.playBeep(250);
            }
        };

        AndroidOS.updateElectroWave = function() {
            const canvas = document.getElementById('electro-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            canvas.width = canvas.parentElement.clientWidth;
            canvas.height = canvas.parentElement.clientHeight;

            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.strokeStyle = '#f59e0b';
            ctx.lineWidth = 2;
            ctx.beginPath();

            const midY = canvas.height / 2;
            for (let x = 0; x < canvas.width; x++) {
                let y = midY + Math.sin(x * 0.15) * 25;
                ctx.lineTo(x, y);
            }
            ctx.stroke();
        };

        AndroidOS.calculateKarvonen = function() {
            const age = parseFloat(document.getElementById('calc-age')?.value) || 0;
            const rest = parseFloat(document.getElementById('calc-rest')?.value) || 0;
            const fcMax = 220 - age;
            const target = Math.round(((fcMax - rest) * 0.6) + rest);

            document.getElementById('calc-result').innerText = `FC Treino (60%): ${target} bpm (FCMáx: ${fcMax})`;
        };

        // Pomodoro logic
        AndroidOS.pomoTime = 1500;
        AndroidOS.pomoTimer = null;

        AndroidOS.startPomo = function() {
            if (this.pomoTimer) return;
            this.pomoTimer = setInterval(() => {
                if (this.pomoTime > 0) {
                    this.pomoTime--;
                    const m = String(Math.floor(this.pomoTime / 60)).padStart(2, '0');
                    const s = String(this.pomoTime % 60).padStart(2, '0');
                    document.getElementById('pomo-clock').innerText = `${m}:${s}`;
                } else {
                    clearInterval(this.pomoTimer);
                    this.pomoTimer = null;
                    this.showToast("Pomodoro", "Tempo de estudo concluído!");
                }
            }, 1000);
        };

        AndroidOS.resetPomo = function() {
            clearInterval(this.pomoTimer);
            this.pomoTimer = null;
            this.pomoTime = 1500;
            document.getElementById('pomo-clock').innerText = '25:00';
        };

        AndroidOS.execCmd = function(cmd) {
            const out = document.getElementById('term-out');
            const input = document.getElementById('term-in');
            if (!out || !input) return;

            const trimmed = cmd.trim().toLowerCase();
            input.value = '';

            const line = document.createElement('div');
            line.innerHTML = `<span class="text-teal-500">$</span> ${trimmed}`;
            out.appendChild(line);

            let res = '';
            if (trimmed === 'ajuda') res = 'Comandos: ajuda, sobre, manguito, limpar';
            else if (trimmed === 'sobre') res = 'RHUTHER OS Android Edition v1.0';
            else if (trimmed === 'manguito') res = 'Supraespinal, Infraespinal, Redondo Menor, Subescapular.';
            else if (trimmed === 'limpar') { out.innerHTML = ''; return; }
            else res = `Comando '${trimmed}' desconhecido.`;

            const resLine = document.createElement('div');
            resLine.className = 'text-slate-300 pl-2';
            resLine.innerText = res;
            out.appendChild(resLine);
        };
